cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(neondoll_routing LANGUAGES C CXX)
    enable_testing()
endif()
if(NOT INTEROP_LWIP_SOURCE_DIR)
    set(INTEROP_LWIP_SOURCE_DIR "${LWIP_SOURCE_DIR}")
endif()
set(LWIP_DIR "${INTEROP_LWIP_SOURCE_DIR}")
include("${LWIP_DIR}/src/Filelists.cmake")
add_library(interop_lwip STATIC ${lwipcore_SRCS} ${lwipcore4_SRCS} ${lwipcore6_SRCS} ${lwipapi_SRCS}
    sys_port.c)
target_include_directories(interop_lwip PUBLIC port "${LWIP_DIR}/src/include" "${LWIP_DIR}/test/unit/arch")
target_compile_options(interop_lwip PRIVATE -ffunction-sections -fdata-sections)
add_executable(interop_routing_tests test_routing.cpp ../bind_probe.cpp)
target_link_libraries(interop_routing_tests PRIVATE interop_lwip)
target_compile_features(interop_routing_tests PRIVATE cxx_std_17)
target_compile_options(interop_routing_tests PRIVATE -UNDEBUG)
target_link_options(interop_routing_tests PRIVATE -Wl,--gc-sections)
add_test(NAME interop_socket_egress COMMAND interop_routing_tests)
